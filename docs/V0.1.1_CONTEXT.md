# Existing Conversation.py

Conversation has been improved.

Current implementation:

```python
from google.genai import types


class Conversation:

    def __init__(self):
        self._history: list[types.Content] = []

    def add_user_message(self, text: str) -> None:
        self._history.append(
            types.Content(role="user", parts=[types.Part(text=text)])
        )

    def add_assistant_message(self, text: str) -> None:
        self._history.append(
            types.Content(role="model", parts=[types.Part(text=text)])
        )

    def rollback_last_message(self) -> None:
        if self._history:
            self._history.pop()

    def get_history(self) -> list[types.Content]:
        return list(self._history)
```

Do NOT remove encapsulation.

Do NOT expose _history directly.

Future versions (SQLite memory) should continue using this class.
